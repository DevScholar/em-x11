cmake_minimum_required(VERSION 3.20)
project(em-x11 C)

# Enforce Emscripten toolchain. The `emcmake` wrapper sets CMAKE_TOOLCHAIN_FILE
# to Emscripten.cmake; if someone runs bare cmake, fail loudly.
if(NOT EMSCRIPTEN)
    message(FATAL_ERROR
        "em-x11 must be configured through emcmake. "
        "From inside WSL / Linux: emcmake cmake -S . -B build")
endif()

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

# Output artifacts into a predictable location that Vite can serve.
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/artifacts")

# Default warning flags for our own code. Third-party code opts out.
set(EMX11_WARN_FLAGS -Wall -Wextra -Wno-unused-parameter)

# Demo helper (emx11_finalize_demo, EMX11_LIB_DIR, EMX11_INCLUDE_DIR).
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(emx11_demo)

add_subdirectory(native)

# Third-party X11 libraries. Their CMakeLists.txt files are copied from
# cmake/third-party/<lib>/ into ignored-area/third-party/<lib>/ by the
# fetch script. config.h is generated by emconfigure ./configure at fetch
# time (no more hand-written overlays).
add_subdirectory(ignored-area/third-party/xtrans)
add_subdirectory(ignored-area/third-party/xorgproto)
add_subdirectory(ignored-area/third-party/libICE)
add_subdirectory(ignored-area/third-party/libSM)
add_subdirectory(ignored-area/third-party/libXt)
add_subdirectory(ignored-area/third-party/libXmu)
add_subdirectory(ignored-area/third-party/libXpm)
add_subdirectory(ignored-area/third-party/libXaw)
add_subdirectory(examples/hello)
add_subdirectory(examples/xt-hello)
add_subdirectory(examples/xcalc)
add_subdirectory(examples/xeyes)
add_subdirectory(examples/glxgears)
# twm wasm artifact needed by twm-session demo.
add_subdirectory(examples/twm)
